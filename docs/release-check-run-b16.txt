// One-off release check: validate tripcurve's SARIF against the official SARIF 2.1.0 JSON schema.
// Run from a scratch folder OUTSIDE the repository, with ajv, ajv-formats and ajv-draft-04 installed there.
// The official schema declares JSON Schema draft-04, which plain ajv 8 does not load; ajv-draft-04 is its
// draft-04 build. usage: node validate.mjs <repo-root> <built-cli.js>
import { readFileSync, readdirSync } from 'node:fs';
import { spawnSync } from 'node:child_process';
import { join } from 'node:path';
import Ajv from 'ajv-draft-04';
import addFormats from 'ajv-formats';

const [repo, cli] = process.argv.slice(2);
const schema = JSON.parse(readFileSync('sarif-schema-2.1.0.json', 'utf8'));
const ajv = new Ajv({ strict: false, allErrors: true });
addFormats(ajv);
const validate = ajv.compile(schema);

let failures = 0;
function check(name, text) {
  let doc;
  try {
    doc = JSON.parse(text);
  } catch (e) {
    console.log(`INVALID JSON  ${name}: ${e.message}`);
    failures += 1;
    return;
  }
  if (validate(doc)) {
    const results = doc.runs.reduce((n, r) => n + (r.results?.length ?? 0), 0);
    const supp = doc.runs.reduce(
      (n, r) => n + (r.results ?? []).filter((x) => (x.suppressions ?? []).length > 0).length, 0);
    console.log(`VALID         ${name} (${results} results, ${supp} with suppressions)`);
  } else {
    failures += 1;
    console.log(`INVALID       ${name}`);
    for (const e of validate.errors.slice(0, 5)) console.log(`    ${e.instancePath || '/'} ${e.message}`);
  }
}

const goldens = readdirSync(join(repo, 'test/golden')).filter((f) => /^report-.*\.sarif\.txt$/.test(f)).sort();
for (const f of goldens) check(`test/golden/${f}`, readFileSync(join(repo, 'test/golden', f), 'utf8'));
for (const dir of ['demo', 'demo-fixed']) {
  const r = spawnSync(process.execPath, [cli, '.', '--format', 'sarif'], { cwd: join(repo, dir), encoding: 'utf8' });
  console.log(`(${dir}: tripcurve exit ${r.status})`);
  check(`${dir}/ (live run, --format sarif)`, r.stdout);
}

// Negative controls: the validator must be able to say no.
const base = JSON.parse(readFileSync(join(repo, 'test/golden/report-waived.sarif.txt'), 'utf8'));
const mutants = {
  'missing version': (d) => { delete d.version; },
  'unknown top-level property': (d) => { d.bogus = 1; },
  'result level not in the enum': (d) => { d.runs[0].results[0].level = 'catastrophe'; },
  'suppression kind not in the enum': (d) => {
    const r = d.runs[0].results.find((x) => (x.suppressions ?? []).length > 0);
    r.suppressions[0].kind = 'maybe';
  },
};
for (const [name, mutate] of Object.entries(mutants)) {
  const d = structuredClone(base);
  mutate(d);
  const ok = validate(d);
  console.log(`${ok ? 'NOT REJECTED (validator cannot fail!)' : 'rejected'}  negative control: ${name}`);
  if (ok) failures += 1;
}
console.log(failures === 0 ? 'RESULT: all documents valid, all negative controls rejected' : `RESULT: ${failures} problem(s)`);
process.exit(failures === 0 ? 0 : 1);
